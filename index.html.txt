    <script>
        function cipherText() {
            let str = document.getElementById("inputText").value;
            // Simple Caesar-like shift for demonstration
            let result = str.split('').map(char => {
                let code = char.charCodeAt(0);
                return String.fromCharCode(code + 3);
            }).join('');
            document.getElementById("outputText").innerText = "Result: " + result;
        }
    </script>
